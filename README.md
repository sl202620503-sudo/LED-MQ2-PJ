# LED-MQ2-PJ
import time
from machine import ADC, Pin

# ----------------------------------------------------
# 1. 핀 설정 (하드웨어 연결)
# ----------------------------------------------------
# MQ2 센서 아날로그 핀 (GP26 = ADC 0번 채널)
mq2_sensor = ADC(Pin(26))

# 3단계 상태 표시 LED 핀 설정
led_green = Pin(15, Pin.OUT)  # 정상 레벨
led_yellow = Pin(14, Pin.OUT)  # 경고 레벨 (초기 오프가스/과열 감지)
led_red = Pin(13, Pin.OUT)  # 위험 레벨 (폭발/화재 임박)

# ----------------------------------------------------
# 2. 임계값(Threshold) 및 예열 시간 설정
# ----------------------------------------------------
# MicroPython ADC의 16비트 읽기 범위: 0 ~ 65535
# *현장 설치 환경에 따라 아래 임계값을 조정하세요.*
WARNING_LEVEL = 18000  # 경고 레벨 (수소/CO 오프가스 감지 시작)
DANGER_LEVEL = 35000  # 위험 레벨 (가스 농도 급증, 폭발 임박)

WARMUP_TIME = 20  # 시연용 예열 시간 (초) - 실제 사용 시 60초 이상 권장


def set_leds(green, yellow, red):
    """LED 상태를 한 번에 제어하는 함수"""
    led_green.value(green)
    led_yellow.value(yellow)
    led_red.value(red)


# ----------------------------------------------------
# 3. 센서 예열 (Preheating) 단계
# ----------------------------------------------------
print("=" * 50)
print("스마트그리드 오프가스 조기 감지 시스템 시작")
print("MQ2 센서 히터 예열 중... 잠시만 기다려주세요.")
print("=" * 50)

# 예열 동안 노란색 LED를 점멸하여 상태 안내
for i in range(WARMUP_TIME):
    set_leds(0, 1, 0)
    time.sleep(0.5)
    set_leds(0, 0, 0)
    time.sleep(0.5)
    print(f"예열 진행 중... {WARMUP_TIME - i}초 남음")

print("\n>> 예열 완료! 실시간 오프가스 모니터링을 시작합니다.\n")

# ----------------------------------------------------
# 4. 실시간 가스 모니터링 루프
# ----------------------------------------------------
try:
    while True:
        # MQ2 센서에서 아날로그 값 읽기 (0 ~ 65535)
        gas_value = mq2_sensor.read_u16()

        # ------------------------------------------------
        # 레벨 1: [정상 레벨]
        # ------------------------------------------------
        if gas_value < WARNING_LEVEL:
            set_leds(1, 0, 0)  # 초록 LED만 ON
            status_msg = " [정상] 설비 안정 상태"

        # ------------------------------------------------
        # 레벨 2: [경고 레벨]
        # ESS 오프가스(수소/CO), 피복 과열, 절연유 분해 초기 감지
        # ------------------------------------------------
        elif WARNING_LEVEL <= gas_value < DANGER_LEVEL:
            set_leds(0, 1, 0)  # 노랑 LED만 ON
            status_msg = " [경고] 오프가스/절연유 분해 전조 감지!"

        # ------------------------------------------------
        # 레벨 3: [위험 레벨]
        # 가스 농도 고위험, 화재/폭발 골든타임 임박
        # ------------------------------------------------
        else:
            set_leds(0, 0, 1)  # 빨강 LED만 ON
            status_msg = " [위험] 폭발/화재 고위험! 즉시 전원 차단 필요!"

        # Thonny 시리얼 플롯터 및 콘솔 출력
        print(f"가스 센서 측정값: {gas_value} | 상태:{status_msg}")

        # 1초 간격으로 측정
        time.sleep(1)

except KeyboardInterrupt:
    # 프로그램 종료 시 모든 LED OFF
    set_leds(0, 0, 0)
    print("\n시스템이 안전하게 종료되었습니다.")
